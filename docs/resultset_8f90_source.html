78 !! statement handle, setting up row status and @ref odbc_columnset::columnset

85 subroutine resultset_new(that, stmt)
86 type(sqlhstmt), intent(in) :: stmt
87 type(resultset), target :: that
88 !private
89 integer(SQLRETURN) :: rc
90
91 that%stmt = stmt
92 that%rows = 0
93 rc = sqlsetstmtattr(that%stmt, sql_attr_row_status_ptr, c_loc(that%status), 0)
94 rc = sqlsetstmtattr(that%stmt, sql_attr_rows_fetched_ptr, c_loc(that%rows), 0)
95 that%rec = 0_sqlsmallint
96 call that%get_metadata(that%columns)
97 end subroutine

99 subroutine resultset_get_metadata(this, columns)
100 class(resultset), intent(inout) :: this
101 type(columnset), intent(inout) :: columns
102 !private
103 integer :: i
104 integer(SQLSMALLINT) :: name_length
105 integer(SQLSMALLINT), allocatable :: column_count
106 type(column), allocatable, target :: cols(:)
107 integer :: nErr, nMsg, iRec, cState
108
109 allocate(column_count, source = 0_sqlsmallint)
110 this%ierr = sqlnumresultcols(this%stmt, column_count)
111 if (this%ierr == sql_error .or. this%ierr == sql_invalid_handle) then
112 call this%handle_errors()
113 end if
114
115 allocate(cols(column_count))
116 do i = 1, column_count
117 allocate(character(51) :: cols(i)%name)
118 cols(i)%decim_size = 0
119 cols(i)%nullable = 0
120 cols(i)%size = 0
121 cols(i)%type = 0
122 this%ierr = sqldescribecol(this%stmt, int(i, c_short), cols(i)%name, len(cols(i)%name, kind=c_short), &
123 name_length, cols(i)%type, cols(i)%size, &
124 cols(i)%decim_size, cols(i)%nullable)
125 if (this%ierr == sql_error .or. this%ierr == sql_invalid_handle) then
126 call this%handle_errors()
127 end if
128 cols(i)%name = clean_string(cols(i)%name)
129 if (allocated(cols(i)%content)) deallocate(cols(i)%content)
130 allocate(character(merge(4096, min(4096, cols(i)%size), cols(i)%size <= 0)) :: cols(i)%content)
131 end do
132 call columns%addrange(cols)
133
134 do i = 1, column_count
135 this%ierr = this%columns%bind(this%stmt, i)
136 if (this%ierr == sql_error) call this%handle_errors()
137 end do
138 end subroutine

251 !> Retrieves a @ref odbc_columnset::column value as an integer by column
252 !! index (1-based) from the @ref resultset.
253 !! @param[inout] this The @ref resultset object.
254 !! @param[in] col The column index (1-based).
255 !! @return The @ref odbc_columnset::column value as an integer, or 0 if the column
256 !! is invalid.
257 !!
258 !! @b Remarks
259 !! @ingroup group_odbc_resultset
260 function resultset_get_integer_from_index(this, col) result(res)
261 class(resultset), intent(inout) :: this
262 integer, intent(in) :: col
263 integer :: res
264 !private
265 type(column), pointer :: c => null()
266 character(:), allocatable :: str
267
268 if (col <= 0 .or. col > this%columns%count()) then
269 res = 0
270 return
271 end if
272
273 c => this%columns%get(col)
274 if (associated(c)) then
275 str = clean_string(c%content)
276 read(str, *) res
277 else
278 res = 0
279 end if
280 nullify(c)
281 end function

495 character(*), intent(in) :: value
496 character(:), allocatable :: str

498 integer i

500 str = value
